# Production-grade async HTTP client with circuit breaker + retry
import asyncio
from contextlib import asynccontextmanager
from dataclasses import dataclass
from typing import Optional, Callable
import aiohttp
import time

@dataclass
class CircuitBreaker:
    failure_threshold: int = 5
    recovery_timeout: float = 30.0
    _failures: int = 0
    _last_failure_time: Optional[float] = None
    _state: str = "CLOSED"

    def can_execute(self) -> bool:
        if self._state == "OPEN":
            if time.time() - self._last_failure_time > self.recovery_timeout:
                self._state = "HALF_OPEN"
                return True
            return False
        return True

    def record_success(self):
        self._failures = 0
        self._state = "CLOSED"

    def record_failure(self):
        self._failures += 1
        self._last_failure_time = time.time()
        if self._failures >= self.failure_threshold:
            self._state = "OPEN"

async def retry_with_backoff(
    fn: Callable, max_retries: int = 3, base_delay: float = 1.0,
    max_delay: float = 60.0, exceptions: tuple = (Exception,)
):
    """Exponential backoff with jitter for resilient API calls."""
    for attempt in range(max_retries + 1):
        try:
            return await fn()
        except exceptions as e:
            if attempt == max_retries:
                raise
            delay = min(base_delay * (2 ** attempt), max_delay)
            jitter = delay * (0.5 + random.random() * 0.5)
            await asyncio.sleep(jitter)

class ResilientWeatherClient:
    def __init__(self, api_key: str):
        self.api_key = api_key
        self._breaker = CircuitBreaker()
        self._session: Optional[aiohttp.ClientSession] = None

    @asynccontextmanager
    async def session(self):
        # Connection pooling + keep-alive for high-throughput
        connector = aiohttp.TCPConnector(
            limit=100, limit_per_host=20,
            ttl_dns_cache=300, use_dns_cache=True
        )
        timeout = aiohttp.ClientTimeout(total=30, connect=5)
        self._session = aiohttp.ClientSession(
            connector=connector, timeout=timeout,
            headers={"Authorization": f"Bearer {self.api_key}"}
        )
        try:
            yield self._session
        finally:
            await self._session.close()
            self._session = None

    async def fetch_forecast(self, lat: float, lon: float) -> dict:
        if not self._breaker.can_execute():
            raise RuntimeError("Circuit breaker is OPEN")

        async def _call():
            url = f"https://api.weather.com/v1/forecast?lat={lat}&lon={lon}"
            async with self._session.get(url) as resp:
                resp.raise_for_status()
                return await resp.json()

        try:
            result = await retry_with_backoff(
                _call, max_retries=3,
                exceptions=(aiohttp.ClientError, asyncio.TimeoutError)
            )
            self._breaker.record_success()
            return result
        except Exception:
            self._breaker.record_failure()
            raise
Decorator — Caching with TTL + Cache Stampede Protection
import functools
import hashlib
import time
from threading import Lock

class TTLCache:
    """Thread-safe TTL cache with probabilistic early expiration to prevent stampede."""
    def __init__(self, ttl: int = 300, beta: float = 1.0):
        self.ttl = ttl
        self.beta = beta  # higher = more aggressive early refresh
        self._cache = {}
        self._lock = Lock()
        self._refreshing = set()  # keys currently being refreshed

    def get(self, key: str, fetch_fn: Callable):
        now = time.time()
        with self._lock:
            if key in self._cache:
                value, expiry = self._cache[key]
                # Probabilistic early expiration (XFetch algorithm)
                delta = now - (expiry - self.ttl)
                ttl_remaining = expiry - now
                if ttl_remaining > 0:
                    # Chance to refresh early increases as expiry approaches
                    prob = exp(-self.beta * ttl_remaining / delta)
                    if random.random() > prob:
                        return value
                # Key expired or chosen for early refresh
                if key in self._refreshing:
                    return value  # Return stale while another thread refreshes
                self._refreshing.add(key)

        # Fetch outside lock
        try:
            new_value = fetch_fn()
            with self._lock:
                self._cache[key] = (new_value, now + self.ttl)
            return new_value
        finally:
            with self._lock:
                self._refreshing.discard(key)

def cached_weather(ttl: int = 300):
    cache = TTLCache(ttl=ttl)
    def decorator(fn):
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            key = hashlib.sha256(
                f"{fn.__name__}:{args}:{kwargs}".encode()
            ).hexdigest()
            return cache.get(key, lambda: fn(*args, **kwargs))
        return wrapper
    return decorator

async def correlation_middleware(request: Request, call_next):
    cid = request.headers.get("X-Correlation-ID", str(uuid.uuid4()))

def process_weather_etl(spark: SparkSession):
    # Read raw data with partition pruning hint
    raw_df = spark.read.format("delta")         .load("s3://weather-bucket/bronze/station-readings")         .filter(col("ingestion_date") >= current_timestamp().cast("date") - lit(7))

    # SALTING: Handle skewed station_id distribution
    # Add random salt (0-9) to skewed keys, then aggregate in 2 phases
    SALT_BUCKETS = 10

    salted_df = raw_df.withColumn(
        "salted_station",
        concat(col("station_id"), lit("_"), (rand() * SALT_BUCKETS).cast("int"))
    )

    # Phase 1: Partial aggregation per salted key
    partial_agg = salted_df.groupBy("salted_station", "date").agg(
        avg("temperature").alias("avg_temp"),
        max("temperature").alias("max_temp"),
        min("temperature").alias("min_temp"),
        count("*").alias("reading_count")
    )

    # Phase 2: Remove salt and final aggregation
    final_agg = partial_agg.withColumn(
        "station_id", regexp_extract(col("salted_station"), "^(.+?)_", 1)
    ).groupBy("station_id", "date").agg(
        avg("avg_temp").alias("avg_temperature"),
        max("max_temp").alias("max_temperature"),
        min("min_temp").alias("min_temperature"),
        sum("reading_count").alias("total_readings")
    )

    # Write to Delta with Z-ORDER for query optimization
    final_agg.write.format("delta")         .mode("merge")         .option("mergeSchema", "true")         .save("s3://weather-bucket/silver/daily-aggregates")

    # Optimize and Z-ORDER for time-series queries
    spark.sql("""
        OPTIMIZE delta.`s3://weather-bucket/silver/daily-aggregates`
        ZORDER BY (station_id, date)
    """)

    # Vacuum old versions (retain 7 days for time travel)
    spark.sql("""
        VACUUM delta.`s3://weather-bucket/silver/daily-aggregates` RETAIN 168 HOURS
    """)

    # Data quality check with constraints
    spark.sql("""
        ALTER TABLE delta.`s3://weather-bucket/silver/daily-aggregates`
        ADD CONSTRAINT valid_temperature CHECK (avg_temperature BETWEEN -100 AND 100)
    """)

async def rate_limit(request: Request, api_key: str = Header(..., alias="X-API-Key")):
    # Tier-based limits: free=100/hr, pro=1000/hr, enterprise=10000/hr
    tier = await redis.hget(f"api:tier:{api_key}", "tier") or "free"
    limits = {"free": 100, "pro": 1000, "enterprise": 10000}
    limit = limits.get(tier, 100)

    key = f"ratelimit:{api_key}:{int(time.time()) // 3600}"
    current = await redis.incr(key)
    if current == 1:
        await redis.expire(key, 3600)

    if current > limit:
        raise HTTPException(
            status_code=429,
            detail={"error": "Rate limit exceeded", "retry_after": await redis.ttl(key)}

    # Add rate limit headers
    request.state.rate_limit_remaining = limit - current
    request.state.rate_limit_reset = await redis.ttl(key)

# Webhook delivery with HMAC signature + retry
async def deliver_webhook(webhook_id: str, event: str, payload: dict):
    wh = await redis.hgetall(f"webhook:{webhook_id}")
    if not wh or wh.get("active") != "1":
        return

    body = json.dumps(payload, separators=(",", ":"))
    timestamp = str(int(time.time()))
    signature = hmac.new(
        wh["secret"].encode(),
        f"{timestamp}.{body}".encode(),
        hashlib.sha256
    ).hexdigest()

    headers = {
        "Content-Type": "application/json",
        "X-Webhook-ID": webhook_id,
        "X-Event-Type": event,
        "X-Signature": f"sha256={signature}",
        "X-Timestamp": timestamp
    }

    async with aiohttp.ClientSession() as session:
        for attempt in range(5):
            try:
                async with session.post(wh["url"], headers=headers, data=body, timeout=10) as resp:
                    if resp.status < 500:
                        return  # Success or 4xx (client error, don't retry)
            except Exception:
                pass
            await asyncio.sleep(2 ** attempt)

    # All retries failed → DLQ
    await redis.lpush("webhook:dlq", json.dumps({"webhook_id": webhook_id, "event": event, "payload": payload}))

        # Few-shot prompt with explicit constraints
        self.prompt = PromptTemplate(
            input_variables=["context", "question"],
            template="""You are a weather data assistant. Use ONLY the provided context to answer.
If the context doesn't contain the answer, say "I don't have enough information about that."
Never make up weather forecasts or data.

Context:
{context}

Question: {question}

Answer concisely in 2-3 sentences:"""
        )
        self.llm = OpenAI(
            openai_api_key=openai_key,
            model_name="gpt-4-turbo-preview",
            temperature=0.1,  # Low creativity for factual tasks
            max_tokens=500,
            streaming=True
        )
        self.qa_chain = LLMChain(llm=self.llm, prompt=self.prompt)

    async def answer(self, question: str) -> RAGResponse:
        import time
        start = time.time()

        # 1. Retrieve relevant documents (top-5)
        docs = self.vectorstore.similarity_search(question, k=5, filter={"category": "faq"})
        context = "

".join([d.page_content for d in docs])

        # 2. Generate answer
        result = await self.qa_chain.ainvoke({"context": context, "question": question})
        answer = result["text"].strip()

        # 3. Validation layer
        if self._contains_hallucination(answer, docs):
            answer = "I don't have enough verified information to answer that accurately."
            confidence = 0.0
        else:
            confidence = self._calculate_confidence(answer, docs)

        # 4. Cost tracking
        tokens = self.llm.get_num_tokens(context + question + answer)

        return RAGResponse(
            answer=answer,
            sources=[{"text": d.page_content[:200], "source": d.metadata.get("source")} for d in docs],
            confidence=confidence,
            tokens_used=tokens,
            latency_ms=(time.time() - start) * 1000
        )

    def _contains_hallucination(self, answer: str, docs: List[Document]) -> bool:
        # Check if answer contains claims not supported by retrieved docs
        doc_text = " ".join([d.page_content.lower() for d in docs])
        # Extract named entities/numbers from answer and verify against docs
        numbers = re.findall(r'\d+(?:\.\d+)?', answer)
        for num in numbers:
            if num not in doc_text:
                return True
        return False

    def _calculate_confidence(self, answer: str, docs: List[Document]) -> float:
        # Simple heuristic: overlap between answer words and doc words
        answer_words = set(answer.lower().split())
        doc_words = set(" ".join([d.page_content for d in docs]).lower().split())
        overlap = len(answer_words & doc_words) / len(answer_words)
        return round(overlap, 2)

    async def stream_answer(self, question: str) -> AsyncGenerator[str, None]:
        # Streaming for better perceived latency
        docs = self.vectorstore.similarity_search(question, k=3)
        context = "

".join([d.page_content for d in docs])

        async for chunk in self.qa_chain.llm.astream({"context": context, "question": question}):
            yield chunk.content

            fn_name = tool_call.function.name
            fn_args = json.loads(tool_call.function.arguments)

                    "content": json.dumps(result)
                })

            model="gpt-4-turbo-preview",
            messages=[
                {"role": "system", "content": "Answer based on the tool results. Be concise."},
                {"role": "user", "content": user_query},
                message,  # assistant's tool call request
                *tool_results
            ]
        )
        return final_response.choices[0].message.content

    # No tool needed — direct answer
    return message.content

// Validation middleware factory
function validateBody<T extends z.ZodTypeAny>(schema: T) {
  return (req: Request, res: Response, next: NextFunction) => {
    const result = schema.safeParse(req.body);
    if (!result.success) {
      return res.status(400).json({
        error: 'Validation failed',
        details: result.error.flatten().fieldErrors
      });
    }
    req.body = result.data;
    next();
  };
}

// Async handler wrapper for clean error handling
function asyncHandler(fn: (req: Request, res: Response, next: NextFunction) => Promise<void>) {
  return (req: Request, res: Response, next: NextFunction) => {
    Promise.resolve(fn(req, res, next)).catch(next);
  };
}
